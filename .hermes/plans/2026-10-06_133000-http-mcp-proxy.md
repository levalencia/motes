# HTTP MCP Proxy Service — Implementation Plan

> **For Hermes:** Use subagent-driven-development skill to implement this plan task-by-task.

**Goal:** Make HTTP MCP servers (like ship.page) work in chat, not just in the Settings test button.

**Problem:** The MCP Python SDK's `streamable_http_client` uses anyio TaskGroups internally. When we call `__aenter__()` manually inside a FastAPI/asyncio request, the anyio cancel scopes corrupt the event loop and crash the entire request. Stdio MCP works because `stdio_client` doesn't use anyio internally.

**Architecture:** Run HTTP MCP connections as a persistent background service (like the background scanner). The service keeps connections alive, discovers tools once, and proxies tool calls from the chat agent loop through an in-memory queue.

```
Chat Request → Agent Loop → MCPProxyTool.execute(args)
                                    ↓
                            asyncio.Queue → MCPProxyService
                                    ↓
                            session.call_tool(name, args) → result
                                    ↓
                            asyncio.Future → back to agent loop
```

**Tech Stack:** Python asyncio, existing MCP SDK, FastAPI app.state

```python
# backend/app/mcp_proxy.py
"""Persistent HTTP MCP proxy service.

Keeps HTTP MCP connections alive in the background using proper async context
managers. Proxies tool calls from the agent loop via asyncio primitives.
"""
from __future__ import annotations

import asyncio
import json
from dataclasses import dataclass, field
from typing import Any

import structlog

logger = structlog.get_logger()


@dataclass
class MCPConnection:
    """A live HTTP MCP connection with its tools."""
    server_name: str
    url: str
    tools: dict[str, Any] = field(default_factory=dict)  # tool_name → tool_schema
    session: Any = None  # MCP ClientSession
    _task: asyncio.Task | None = None


class MCPProxyService:
    """Background service managing persistent HTTP MCP connections."""

    def __init__(self) -> None:
        self._connections: dict[str, MCPConnection] = {}
        self._running = False
        self._lock = asyncio.Lock()

    def is_connected(self, server_name: str) -> bool:
        return server_name in self._connections

    def get_tool_schemas(self, server_name: str) -> dict[str, Any]:
        """Return discovered tool schemas for a connected server."""
        conn = self._connections.get(server_name)
        return conn.tools if conn else {}

    def all_tool_schemas(self) -> dict[str, dict[str, Any]]:
        """Return all tools across all connected servers."""
        result = {}
        for name, conn in self._connections.items():
            for tool_name, schema in conn.tools.items():
                result[f"mcp_{name}__{tool_name}"] = schema
        return result

    async def connect(self, server_name: str, url: str) -> list[str]:
        """Connect to an HTTP MCP server. Returns list of tool names."""
        if server_name in self._connections:
            return list(self._connections[server_name].tools.keys())

        conn = MCPConnection(server_name=server_name, url=url)
        conn._task = asyncio.create_task(
            self._run_connection(conn),
            name=f"mcp_proxy_{server_name}",
        )
        # Wait for connection to establish (up to 15s)
        for _ in range(30):
            await asyncio.sleep(0.5)
            if conn.session is not None:
                self._connections[server_name] = conn
                logger.info("mcp_proxy_connected", server=server_name, tools=len(conn.tools))
                return list(conn.tools.keys())
            if conn._task.done():
                break

        logger.warning("mcp_proxy_connect_timeout", server=server_name)
        return []

    async def _run_connection(self, conn: MCPConnection) -> None:
        """Run a persistent MCP connection inside proper async with."""
        try:
            from mcp import ClientSession
            from mcp.client.streamable_http import streamable_http_client

            async with streamable_http_client(conn.url) as transport:  # noqa: SIM117
                async with ClientSession(transport[0], transport[1]) as session:
                    await session.initialize()
                    conn.session = session

                    # Discover tools
                    result = await session.list_tools()
                    for t in result.tools:
                        conn.tools[t.name] = {
                            "name": t.name,
                            "description": t.description or "",
                            "inputSchema": t.inputSchema if hasattr(t, "inputSchema") else {},
                        }

                    # Keep alive — wait until cancelled
                    while True:
                        await asyncio.sleep(60)

        except asyncio.CancelledError:
            logger.info("mcp_proxy_disconnected", server=conn.server_name)
        except BaseException as exc:
            logger.warning("mcp_proxy_error", server=conn.server_name, error=str(exc))
        finally:
            conn.session = None

    async def call_tool(self, server_name: str, tool_name: str, arguments: dict) -> dict:
        """Call a tool on a connected HTTP MCP server."""
        conn = self._connections.get(server_name)
        if not conn or not conn.session:
            return {"error": f"MCP server '{server_name}' not connected"}

        try:
            result = await conn.session.call_tool(tool_name, arguments=arguments)
            texts = []
            for item in result.content:
                if hasattr(item, "text"):
                    texts.append(item.text)
            return {"result": "\n".join(texts) if texts else str(result)}
        except BaseException as exc:
            logger.warning("mcp_proxy_call_failed", server=server_name, tool=tool_name, error=str(exc))
            return {"error": f"Tool call failed: {exc}"}

    async def disconnect(self, server_name: str | None = None) -> None:
        """Disconnect from one or all servers."""
        names = [server_name] if server_name else list(self._connections.keys())
        for name in names:
            conn = self._connections.pop(name, None)
            if conn and conn._task:
                conn._task.cancel()

    async def disconnect_all(self) -> None:
        """Disconnect all servers."""
        await self.disconnect()
```

```python
class MCPProxyTool(Tool):
    """Tool that proxies calls through MCPProxyService for HTTP MCP servers."""

    def __init__(self, tool_name, server_name, tool_description, tool_parameters, proxy):
        self._tool_name = tool_name
        self._server_name = server_name
        self._description = tool_description
        self._parameters = tool_parameters
        self._proxy = proxy

    @property
    def name(self):
        safe = re.sub(r"[^a-zA-Z0-9]", "_", self._server_name).strip("_").lower()
        return f"mcp_{safe}__{self._tool_name}"

    @property
    def description(self):
        return self._description

    @property
    def parameters(self):
        return self._parameters

    async def execute(self, arguments):
        result = await self._proxy.call_tool(self._server_name, self._tool_name, arguments)
        return json.dumps(result)
```

## Risks & Tradeoffs

1. **Connection lifetime:** HTTP MCP connections may timeout after inactivity. The `_run_connection` loop pings every 60s to keep alive. May need reconnect logic.
2. **Multi-user:** Currently one proxy service for all users. If users add different MCP servers, they share the proxy. May need per-user isolation later.
3. **Memory:** Each connection holds a session object in memory. 10 servers = 10 sessions. Acceptable for self-hosted.
4. **Startup time:** Proxy connects at app startup. If ship.page is slow (10s), it delays readiness. Mitigated by connecting in background tasks.
